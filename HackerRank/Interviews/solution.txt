    colleges.contest_id,
    contests.hacker_id,
    contests.name,

    SUM(sub3.total_accepted_submissions) AS total_accepted_submissions,
    SUM(sub3.total_views) AS total_views,
    SUM(sub3.total_unique_views) AS total_unique_views

    SELECT
        c.challenge_id,
